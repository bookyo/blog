Five lines, no dependencies, every checkout

This function has been pasted into more JavaScript files than any other arithmetic routine. The whole algorithm lives in alt = !alt.

function luhnValid(num) {
  const digits = num.replace(/\D/g, '').split('').map(Number);
  if (digits.length < 12) return false;
  let sum = 0, alt = false;
  for (let i = digits.length - 1; i >= 0; i--) {
    let d = digits[i];
    if (alt) { d *= 2; if (d > 9) d -= 9; }
    sum += d; alt = !alt;
  }
  return sum % 10 === 0;
}

Filed in 1960. The same five lines, in the same order, in every language, on every checkout form on earth.
