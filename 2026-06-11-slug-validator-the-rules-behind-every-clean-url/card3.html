function isValidSlug(s, opts={}) {
  const allow = opts.allowUnderscores ?? false;
  const max   = opts.maxLength ?? 200;
  if (!s || s.length > max)        return false;
  if (s.startsWith('-') ||
      s.endsWith('-'))                return false;
  if (s.includes('--'))               return false;
  const re = allow
    ? /^[a-z0-9_-]+$/
    : /^[a-z0-9-]+$/;
  return re.test(s);
}
